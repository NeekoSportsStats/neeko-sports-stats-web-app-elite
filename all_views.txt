=== schema=public | view=v_mw_best_trades ===
 WITH buy_players AS (

            v_mw_premium.player_name,

        ), sell_players AS (
         SELECT v_mw_premium.player_id,
            v_mw_premium.player_name,
            v_mw_premium.team,

            v_mw_premium.price,
